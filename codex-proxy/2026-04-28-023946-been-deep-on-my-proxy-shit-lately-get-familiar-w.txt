  // CODEX_PROXY_HOOK_MODULE=/Users/andrejfidanovski/projects/codex-proxy/hooks/
  profiles.ts
  import profilesConfig from
  "/Users/andrejfidanovski/.codex-proxy/profiles.json";

  export const hooks = {
    onRequest(body, ctx) {
      const profile = identifyProfile(ctx.headers, profilesConfig);

      // 1. Demote any input[] item where role > writeRoles[max]
      body.input = body.input.map((m) => {
        const tier = m.role || m.type;
        if (!profile.writeRoles.includes(tier)) {
          // teammate writing developer → rewrite to user
          return { ...m, role: "user", type: "message" };
        }
        return m;
      });

      // 2. Mandatory developer prefix (uncontestable — proxy injects)
      if (profile.developerPrefix) {
        body.input.unshift({
          role: "developer",
          content: [{ type: "input_text", text: profile.developerPrefix }],
        });
      }

      // 3. Tool gate
      if (profile.tools !== "*") {
        body.tools = (body.tools || []).filter((t) =>
          profile.tools.includes(t.name) &&
  !profile.blockedTools?.includes(t.name)
        );
      }

      // 4. Content cap on user-tier text
      if (profile.contentMaxBytes) capUserContent(body,
  profile.contentMaxBytes);

      // 5. Rate limit (separate counter store)
      if (rateLimitExceeded(profile, ctx)) throw new Error("rate_limited");

      // Tag for downstream observability
      body.client_metadata = { ...body.client_metadata, "x-codex-proxy-profile":
   profile.name };
      return body;
    },

    onSseEvent(event, ctx) {
      const profile = profileFromRequestId(ctx.requestId);
      if (ctx.channel && !profile.readChannels.includes(ctx.channel)) {
        return "drop";  // teammate never sees analysis
      }
      return null;
    },
  };

  developer-tier, it sits ABOVE any user-tier prompt-injection attempt. The
  model's training honors that ordering. Anti-jailbreak by trust-hierarchy.
